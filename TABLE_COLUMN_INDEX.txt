COLUMNS:
────────────────────────────────────────────────────────────────────────────
 # │ Column         │ Type        │ Constraints                  │ Index
───┼────────────────┼─────────────┼──────────────────────────────┼────────
 1 │ id             │ UUID        │ PK, Default: gen_random_uuid │ YES
 2 │ student_id     │ UUID        │ FK → users, NOT NULL         │ YES
 3 │ course_id      │ UUID        │ FK → courses, NOT NULL       │ YES
 4 │ status         │ TEXT        │ NOT NULL, CHECK, Default     │ YES
